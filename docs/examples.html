03 - callbacks

Async callbacks

Repo: examples/03 - callbacks

import random
import glue

glue.init()

@glue.expose
def py_random():
    return random.random()

@glue.expose
def py_exception(error):
    if error:
        raise ValueError("Test")
    return "No Error"

def print_num(n):
    print('Got this from Javascript:', n)

def print_num_failed(error, stack):
    print("Error:", error)
    print("Stack:", stack)

glue.js_random()(print_num)
glue.js_random()(lambda n: print('Got this from Javascript:', n))
glue.js_with_error()(print_num, print_num_failed)

glue.start('callbacks.html')

08 - createreactapp

React + TypeScript

Repo: examples/08 - createreactapp

import os
import random
import sys
import glue

@glue.expose
def say_hello_py(x):
    print('Hello from %s' % x)
    glue.say_hello_js('Python!')

@glue.expose
def pick_file(folder):
    folder = os.path.expanduser(folder)
    if not os.path.isdir(folder):
        return '%s is not a valid folder' % folder
    files = [f for f in os.listdir(folder)
             if not os.path.isdir(os.path.join(folder, f))]
    return random.choice(files) if files else 'No files'

def start_glue(develop):
    if develop:
        glue.init('src', ['.tsx', '.ts', '.jsx', '.js', '.html'])
        page, mode = {'port': 3000}, None
    else:
        glue.init('build', ['.tsx', '.ts', '.jsx', '.js', '.html'])
        page, mode = 'index.html', 'auto'
    glue.start(page, mode=mode, host='localhost', port=8080)

if __name__ == '__main__':
    start_glue(develop=len(sys.argv) == 2)
